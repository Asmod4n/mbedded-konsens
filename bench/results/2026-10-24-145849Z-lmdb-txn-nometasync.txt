#include <array>

#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <cstring>

#include <stdexcept>

#include <lmdb.h>

#ifndef ARM
#error ARM
#endif

namespace {

constexpr int arm_step = 1;
constexpr int arm_message_sync = 2;
constexpr int arm_nested = 3;
constexpr int arm_nosync_then_env_sync = 4;
constexpr int arm_writemap_step = 5;
constexpr int arm_no_reads = 6;
constexpr int arm_nometasync_pipelined = 7;
constexpr int steps_per_sample = 10;
constexpr std::size_t map_size = std::size_t{8} << 30;

void check(int rc, const char* what)
{
    if (rc != 0) [[unlikely]] {
        std::fprintf(stderr, "%s: %s\n", what, mdb_strerror(rc));
        std::exit(1);
    }
}

struct stores {
    MDB_env* env;
    MDB_dbi log;
    MDB_dbi meta;
};

std::array<unsigned char, 8> big_endian(std::uint64_t v)
{
    std::array<unsigned char, 8> b{};
    for (int i = 7; i >= 0; --i) {
        b[static_cast<std::size_t>(i)] = static_cast<unsigned char>(v & 0xff);
        v >>= 8;
    }
    return b;
}

std::uint64_t from_big_endian(const MDB_val& v)
{
    std::uint64_t r = 0;
    const auto* p = static_cast<const unsigned char*>(v.mv_data);
    for (std::size_t i = 0; i < v.mv_size; ++i) {
        r = (r << 8) | p[i];
    }
    return r;
}

std::uint64_t read_u64(MDB_txn* txn, MDB_dbi dbi, const char* key)
{
    MDB_val k{std::strlen(key), const_cast<char*>(key)};
    MDB_val v;
    const int rc = mdb_get(txn, dbi, &k, &v);
    if (rc == MDB_NOTFOUND) {
        return 0;
    }
    check(rc, "mdb_get");
    return from_big_endian(v);
}

void write_u64(MDB_txn* txn, MDB_dbi dbi, const char* key, std::uint64_t value)
{
    auto b = big_endian(value);
    MDB_val k{std::strlen(key), const_cast<char*>(key)};
    MDB_val v{b.size(), b.data()};
    check(mdb_put(txn, dbi, &k, &v, 0), "mdb_put meta");
}

struct durable_values {
    std::uint64_t promise;
    std::uint64_t accepted_round;
    std::uint64_t decided;
    std::uint64_t log_len;
};

durable_values read_values(MDB_txn* txn, const stores& s)
{
    durable_values d{read_u64(txn, s.meta, "promise"), read_u64(txn, s.meta, "accepted_round"),
                     read_u64(txn, s.meta, "decided"), 0};
    MDB_cursor* c;
    check(mdb_cursor_open(txn, s.log, &c), "mdb_cursor_open");
    MDB_val k, v;
    const int rc = mdb_cursor_get(c, &k, &v, MDB_LAST);
    mdb_cursor_close(c);
    MDB_stat st;
    check(mdb_stat(txn, s.log, &st), "mdb_stat");
    if (rc == MDB_NOTFOUND) {
        d.log_len = 0;
    } else {
        check(rc, "mdb_cursor_get");
        d.log_len = from_big_endian(k) + 1;
    }
    if (d.log_len != st.ms_entries) [[unlikely]] {
        std::fprintf(stderr, "continuity\n");
        std::exit(1);
    }
    return d;
}

void handle_message(MDB_txn* txn, const stores& s, std::span<unsigned char> payload, durable_values& local)
{
    durable_values d = local;
    if constexpr (ARM != arm_no_reads) {
        d = read_values(txn, s);
    }
    if (d.promise < 1) {
        write_u64(txn, s.meta, "promise", 1);
        write_u64(txn, s.meta, "accepted_round", 1);
    }
    auto key = big_endian(d.log_len);
    MDB_val k{key.size(), key.data()};
    MDB_val v{payload.size(), payload.data()};
    check(mdb_put(txn, s.log, &k, &v, MDB_APPEND), "mdb_put log");
    write_u64(txn, s.meta, "decided", d.log_len);
    local = {1, 1, d.log_len, d.log_len + 1};
}

void run_step(const stores& s, int messages, std::span<unsigned char> payload, durable_values& local)
{
    if constexpr (ARM == arm_message_sync || ARM == arm_nosync_then_env_sync) {
        for (int m = 0; m < messages; ++m) {
            MDB_txn* txn;
            check(mdb_txn_begin(s.env, nullptr, ARM == arm_nosync_then_env_sync ? MDB_NOSYNC : 0, &txn), "begin");
            handle_message(txn, s, payload, local);
            check(mdb_txn_commit(txn), "commit");
        }
        if constexpr (ARM == arm_nosync_then_env_sync) {
            check(mdb_env_sync(s.env, 1), "mdb_env_sync");
        }
    } else {
        MDB_txn* txn;
        check(mdb_txn_begin(s.env, nullptr, ARM == arm_nometasync_pipelined ? MDB_NOMETASYNC : 0, &txn), "begin");
        for (int m = 0; m < messages; ++m) {
            if constexpr (ARM == arm_nested) {
                MDB_txn* child;
                check(mdb_txn_begin(s.env, txn, 0, &child), "begin child");
                handle_message(child, s, payload, local);
                check(mdb_txn_commit(child), "commit child");
            } else {
                handle_message(txn, s, payload, local);
            }
        }
        check(mdb_txn_commit(txn), "commit");
    }
}

int main(int argc, char** argv)
{
    if (argc != 5) {
        std::fprintf(stderr, "usage: bench_txn <samples> <messages_per_step> <payload_bytes> <dir>\n");
        return 1;
    }
    const int samples = std::atoi(argv[1]);
    const int messages = std::atoi(argv[2]);
    const auto payload_bytes = static_cast<std::size_t>(std::atoi(argv[3]));
    stores s{};
    check(mdb_env_create(&s.env), "mdb_env_create");
    check(mdb_env_set_mapsize(s.env, map_size), "mapsize");
    check(mdb_env_set_maxdbs(s.env, 2), "maxdbs");
    const unsigned env_flags = MDB_NOTLS | (ARM == arm_writemap_step ? MDB_WRITEMAP : 0);
    check(mdb_env_open(s.env, argv[4], env_flags, 0600), "mdb_env_open");
    MDB_txn* txn;
    check(mdb_txn_begin(s.env, nullptr, 0, &txn), "begin");
    check(mdb_dbi_open(txn, "log", MDB_CREATE, &s.log), "dbi log");
    check(mdb_dbi_open(txn, "meta", MDB_CREATE, &s.meta), "dbi meta");
    check(mdb_txn_commit(txn), "commit");
    std::vector<unsigned char> payload(payload_bytes, 0x5a);
    durable_values local{};
    run_step(s, messages, payload, local);
    std::vector<double> ns;
    for (int i = 0; i < samples; ++i) {
        const auto t0 = std::chrono::steady_clock::now();
        for (int st = 0; st < steps_per_sample; ++st) {
            run_step(s, messages, payload, local);
        }
        if constexpr (ARM == arm_nometasync_pipelined) {
            check(mdb_env_sync(s.env, 1), "mdb_env_sync");
        }
        const auto t1 = std::chrono::steady_clock::now();
        ns.push_back(std::chrono::duration<double, std::nano>(t1 - t0).count() /
                     (static_cast<double>(steps_per_sample) * messages));
    }
    std::ranges::sort(ns);
    MDB_txn* rt;
    check(mdb_txn_begin(s.env, nullptr, MDB_RDONLY, &rt), "begin ro");
    const auto d = read_values(rt, s);
    mdb_txn_abort(rt);
    std::printf("arm=%d messages_per_step=%d payload_bytes=%zu samples=%d ns_per_message_median=%.1f min=%.1f max=%.1f "
                "log_len=%llu\n",
                ARM, messages, payload_bytes, samples, ns[ns.size() / 2], ns.front(), ns.back(),
                static_cast<unsigned long long>(d.log_len));
    mdb_env_close(s.env);
}
