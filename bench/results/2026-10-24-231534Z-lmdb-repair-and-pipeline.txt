# LMDB repair from a peer, and the pipeline before the decide
#
# Run 2026-10-24 23:15:34 UTC in podman localhost/cxx-matrix on the session machine.
# The code of the bench is not in the tree. It stands at the end of this file.
# Commit of the plan: lmdb-only-source 90fbc9c, clean.
#
# What is measured.
# Bench "repair": one LMDB env of n entries, 8-byte big-endian keys, 1 KiB values,
#   checksum 8 bytes, MDB_NOTLS | MDB_REMAP_CHUNKS. The second branch page is damaged.
#   Arm copy: mdb_env_copyfd2 without flag on the peer over a socketpair, fdatasync,
#     read of every entry, rename. bytes = bytes on the socket.
#   Arm rebuild: new env from the own readable entries plus the missing range from the
#     peer (binary search for the next readable index), fdatasync, rename.
#   seconds = wall time of the whole repair, peer and receiver in one process.
# Bench "pipeline": two envs (log, state), each with checksum, REMAP_CHUNKS, NOTLS.
#   Requests arrive open loop at rate per second for 2 s. The decide of a batch is
#   simulated as send time + rtt_us. Each batch is one sync commit of the log.
#   Every arm executes each request in a child txn of one open state txn.
#   Arm 1: one batch in flight before the decide.
#   Arm 2: up to 64 batches in flight. On a decide that covers only part of the
#     batches: abort the state txn, execute the decided batches again, commit, then
#     execute the batches that are not decided again (one replay).
#   Arm 3: up to 2 batches in flight; the state commits only when every executed
#     batch is decided.
#   The execution of a request is one get and one put of an 8-byte counter. A replay costs more where the
#   execution costs more; that is not measured here.
#   latency = arrival to reply, reply after the decide and the state commit.
# Each arm is its own binary, built twice (copy a, copy b) as the A/A pair. The A/A
# difference is the noise floor. 5 rounds, order of builds, arms and copies reversed
# in every second round. Medians are per binary.
# Builds: gcc-16 / g++-16 16.2.0 -O2 -DNDEBUG, clang-22 / clang++-22 22.1.8 -O2 -DNDEBUG,
#   -std=c++26 -Wall -Wextra -Werror, mdb.c and midl.c as C objects of the same compiler.

static void fnv(const MDB_val* src, MDB_val* dst, const MDB_val*)
{
    std::uint64_t h = 1469598103934665603ull;
    auto p = static_cast<const unsigned char*>(src->mv_data);
    for (std::size_t i = 0; i < src->mv_size; ++i) { h ^= p[i]; h *= 1099511628211ull; }
    std::memcpy(dst->mv_data, &h, 8);
}

constexpr unsigned page_size = 4096;
constexpr unsigned flags_offset = 18;
constexpr std::size_t val_size = 1024;
constexpr std::size_t map_size = 1ull << 30;

static void die(const char* what, int rc) { std::fprintf(stderr, "%s: %s\n", what, mdb_strerror(rc)); std::exit(1); }

static MDB_env* open_env(const std::string& dir)
{
    MDB_env* env; mdb_env_create(&env);
    mdb_env_set_checksum(env, fnv, 8);
    mdb_env_set_mapsize(env, map_size);
    if (int rc = mdb_env_open(env, dir.c_str(), MDB_NOTLS | MDB_REMAP_CHUNKS, 0644)) die("open", rc);
    return env;
}

static void be64(std::uint64_t v, unsigned char* out) { for (int i = 7; i >= 0; --i) { out[i] = v & 0xff; v >>= 8; } }
static std::uint64_t rd64(const unsigned char* p) { std::uint64_t v = 0; for (int i = 0; i < 8; ++i) v = (v << 8) | p[i]; return v; }

static void fill(const std::string& dir, std::uint64_t n)
{
    std::filesystem::remove_all(dir); std::filesystem::create_directory(dir);
    MDB_env* env = open_env(dir);
    MDB_txn* t; mdb_txn_begin(env, nullptr, 0, &t);
    MDB_dbi d; mdb_dbi_open(t, nullptr, 0, &d);
    std::vector<unsigned char> v(val_size, 'x'); unsigned char k[8];
    for (std::uint64_t i = 0; i < n; ++i) {
        be64(i, k); std::memcpy(v.data(), k, 8);
        MDB_val kk{8, k}, vv{v.size(), v.data()};
        if (int rc = mdb_put(t, d, &kk, &vv, MDB_APPEND)) die("put", rc);
    }
    if (int rc = mdb_txn_commit(t)) die("commit", rc);
    mdb_env_close(env);
}

static void damage_branch(const std::string& file)
{
    std::fstream f(file, std::ios::in | std::ios::out | std::ios::binary);
    std::vector<char> pg(page_size);
    std::size_t branch = 0, seen = 0;
    for (std::size_t p = 2;; ++p) {
        f.seekg(p * page_size); if (!f.read(pg.data(), page_size)) break;
        std::uint16_t fl; std::memcpy(&fl, pg.data() + flags_offset, 2);
        if (fl & 1) { ++seen; if (seen == 2) { branch = p; break; } }
    }
    std::size_t off = branch * page_size + page_size / 2;
    f.clear(); f.seekg(off); char c; f.read(&c, 1); c ^= 0x5a; f.seekp(off); f.write(&c, 1);
}

[[maybe_unused]] static int get_fresh(MDB_env* env, MDB_dbi d, std::uint64_t i)
{
    MDB_txn* r; mdb_txn_begin(env, nullptr, MDB_RDONLY, &r);
    unsigned char k[8]; be64(i, k); MDB_val kk{8, k}, vv;
    int rc = mdb_get(r, d, &kk, &vv);
    mdb_txn_abort(r);
    return rc;
}

static std::uint64_t verify(MDB_env* env, std::uint64_t n)
{
    MDB_txn* r; mdb_txn_begin(env, nullptr, MDB_RDONLY, &r);
    MDB_dbi d; mdb_dbi_open(r, nullptr, 0, &d);
    MDB_cursor* c; mdb_cursor_open(r, d, &c);
    MDB_val k, v; std::uint64_t cnt = 0;
    int rc = mdb_cursor_get(c, &k, &v, MDB_FIRST);
    while (rc == 0) {
        if (rd64(static_cast<unsigned char*>(k.mv_data)) != cnt) { std::fprintf(stderr, "gap at %llu\n", (unsigned long long)cnt); std::exit(1); }
        ++cnt; rc = mdb_cursor_get(c, &k, &v, MDB_NEXT);
    }
    if (rc != MDB_NOTFOUND) die("verify", rc);
    mdb_cursor_close(c); mdb_txn_abort(r);
    if (cnt != n) { std::fprintf(stderr, "count %llu\n", (unsigned long long)cnt); std::exit(1); }
    return cnt;
}

[[maybe_unused]] static std::size_t read_all(int fd, std::vector<unsigned char>& out)
{
    unsigned char b[1 << 16]; std::size_t total = 0; ssize_t r;
    while ((r = read(fd, b, sizeof b)) > 0) { out.insert(out.end(), b, b + r); total += r; }
    return total;
}

struct result { double seconds; std::size_t bytes; };

[[maybe_unused]] static result arm_copy(MDB_env* peer, const std::string& dir, std::uint64_t n)
{
    auto t0 = std::chrono::steady_clock::now();
    int sv[2]; socketpair(AF_UNIX, SOCK_STREAM, 0, sv);
    std::thread sender([&] { if (int rc = mdb_env_copyfd2(peer, sv[0], 0)) die("copyfd2", rc); close(sv[0]); });
    std::string ndir = dir + "-new"; std::filesystem::remove_all(ndir); std::filesystem::create_directory(ndir);
    std::FILE* out = std::fopen((ndir + "/data.mdb").c_str(), "wb");
    unsigned char b[1 << 16]; std::size_t bytes = 0; ssize_t r;
    while ((r = read(sv[1], b, sizeof b)) > 0) { std::fwrite(b, 1, r, out); bytes += r; }
    std::fflush(out); fdatasync(fileno(out)); std::fclose(out); close(sv[1]); sender.join();
    MDB_env* e = open_env(ndir);
    verify(e, n);
    mdb_env_close(e);
    std::filesystem::rename(ndir + "/data.mdb", dir + "/data.mdb");
    auto t1 = std::chrono::steady_clock::now();
    return {std::chrono::duration<double>(t1 - t0).count(), bytes};
}

[[maybe_unused]] static void send_range(MDB_env* peer, int fd, std::uint64_t from, std::uint64_t to)
{
    MDB_txn* r; mdb_txn_begin(peer, nullptr, MDB_RDONLY, &r);
    MDB_dbi d; mdb_dbi_open(r, nullptr, 0, &d);
    MDB_cursor* c; mdb_cursor_open(r, d, &c);
    unsigned char k0[8]; be64(from, k0); MDB_val k{8, k0}, v;
    int rc = mdb_cursor_get(c, &k, &v, MDB_SET_RANGE);
    while (rc == 0 && rd64(static_cast<unsigned char*>(k.mv_data)) < to) {
        if (write(fd, k.mv_data, 8) != 8 || write(fd, v.mv_data, v.mv_size) != static_cast<ssize_t>(v.mv_size)) std::exit(1);
        rc = mdb_cursor_get(c, &k, &v, MDB_NEXT);
    }
    mdb_cursor_close(c); mdb_txn_abort(r);
}

[[maybe_unused]] static result arm_rebuild(MDB_env* peer, const std::string& dir, std::uint64_t n)
{
    auto t0 = std::chrono::steady_clock::now();
    MDB_env* own = open_env(dir);
    MDB_dbi od;
    { MDB_txn* r; mdb_txn_begin(own, nullptr, MDB_RDONLY, &r); mdb_dbi_open(r, nullptr, 0, &od); mdb_txn_abort(r); }
    std::string ndir = dir + "-new"; std::filesystem::remove_all(ndir); std::filesystem::create_directory(ndir);
    MDB_env* ne = open_env(ndir);
    MDB_txn* w; mdb_txn_begin(ne, nullptr, 0, &w);
    MDB_dbi nd; mdb_dbi_open(w, nullptr, 0, &nd);
    std::size_t bytes = 0;
    std::uint64_t next = 0;
    while (next < n) {
        MDB_txn* r; mdb_txn_begin(own, nullptr, MDB_RDONLY, &r);
        MDB_cursor* c; mdb_cursor_open(r, od, &c);
        unsigned char k0[8]; be64(next, k0); MDB_val k{8, k0}, v;
        int rc = mdb_cursor_get(c, &k, &v, MDB_SET_RANGE);
        while (rc == 0) {
            std::uint64_t idx = rd64(static_cast<unsigned char*>(k.mv_data));
            if (idx != next) break;
            if (int prc = mdb_put(w, nd, &k, &v, MDB_APPEND)) die("rebuild put", prc);
            ++next; rc = mdb_cursor_get(c, &k, &v, MDB_NEXT);
        }
        mdb_cursor_close(c); mdb_txn_abort(r);
        if (next >= n) break;
        std::uint64_t lo = next, hi = n;
        while (lo < hi) {
            std::uint64_t mid = lo + (hi - lo) / 2;
            bool ok = get_fresh(own, od, mid) == 0;
            bool prev_ok = mid == next ? false : get_fresh(own, od, mid - 1) == 0;
            if (ok && !prev_ok) { lo = mid; break; }
            if (ok) hi = mid; else lo = mid + 1;
        }
        std::uint64_t until = lo;
        int sv[2]; socketpair(AF_UNIX, SOCK_STREAM, 0, sv);
        std::thread sender([&] { send_range(peer, sv[0], next, until); close(sv[0]); });
        std::vector<unsigned char> buf; bytes += read_all(sv[1], buf); close(sv[1]); sender.join();
        for (std::size_t o = 0; o + 8 + val_size <= buf.size(); o += 8 + val_size) {
            MDB_val k2{8, buf.data() + o}, v2{val_size, buf.data() + o + 8};
            if (int prc = mdb_put(w, nd, &k2, &v2, MDB_APPEND)) die("peer put", prc);
        }
        next = until;
    }
    if (int rc = mdb_txn_commit(w)) die("rebuild commit", rc);
    mdb_env_close(own);
    verify(ne, n);
    mdb_env_close(ne);
    std::filesystem::rename(ndir + "/data.mdb", dir + "/data.mdb");
    auto t1 = std::chrono::steady_clock::now();
    return {std::chrono::duration<double>(t1 - t0).count(), bytes};
}

int main(int, char** argv)
{
    std::uint64_t n = std::strtoull(argv[1], nullptr, 10);
    std::string base = argv[2];
    std::string peerdir = base + "/peer", owndir = base + "/own";
    if (!std::filesystem::exists(peerdir + "/data.mdb")) fill(peerdir, n);

#if ARM_COPY

#include <lmdb.h>
#include <algorithm>
#include <chrono>
#include <cstdint>
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <deque>

#include <string>
#include <thread>
#include <vector>

using clk = std::chrono::steady_clock;

constexpr std::size_t map_size = 1ull << 32;
constexpr std::size_t log_entry_bytes = 64;
constexpr std::uint32_t state_keys = 1024;
constexpr std::size_t window_batches = 2;
constexpr std::size_t max_in_flight = 64;

static void fnv(const MDB_val* src, MDB_val* dst, const MDB_val*)
{
    std::uint64_t h = 1469598103934665603ull;
    auto p = static_cast<const unsigned char*>(src->mv_data);
    for (std::size_t i = 0; i < src->mv_size; ++i) { h ^= p[i]; h *= 1099511628211ull; }
    std::memcpy(dst->mv_data, &h, 8);
}

static void check(int rc, const char* what) { if (rc) { std::fprintf(stderr, "%s: %s\n", what, mdb_strerror(rc)); std::exit(1); } }

static MDB_env* open_env(const std::string& dir)
{
    std::filesystem::create_directories(dir);
    MDB_env* env; check(mdb_env_create(&env), "create");
    check(mdb_env_set_checksum(env, fnv, 8), "checksum");
    check(mdb_env_set_mapsize(env, map_size), "mapsize");
    check(mdb_env_open(env, dir.c_str(), MDB_NOTLS | MDB_REMAP_CHUNKS, 0644), "open");
    return env;
}

struct batch { std::uint64_t first; std::uint64_t count; clk::time_point ack; };

struct node {
    MDB_env* log_env;
    MDB_env* state_env;
    MDB_dbi log_dbi;
    MDB_dbi state_dbi;
    std::uint64_t log_len = 0;
    MDB_txn* state_txn = nullptr;
};

static void append_log(node& n, const batch& b)
{
    MDB_txn* t; check(mdb_txn_begin(n.log_env, nullptr, 0, &t), "log begin");
    unsigned char k[8], v[log_entry_bytes] = {};
    for (std::uint64_t i = 0; i < b.count; ++i) {
        std::uint64_t idx = n.log_len++;
        for (int j = 7; j >= 0; --j) k[7 - j] = static_cast<unsigned char>(idx >> (8 * j));
        std::memcpy(v, &idx, 8);
        MDB_val kk{8, k}, vv{sizeof v, v};
        check(mdb_put(t, n.log_dbi, &kk, &vv, MDB_APPEND), "log put");
    }
    check(mdb_txn_commit(t), "log commit");
}

static void execute(node& n, const batch& b)
{
    for (std::uint64_t i = 0; i < b.count; ++i) {
        std::uint32_t key = static_cast<std::uint32_t>((b.first + i) % state_keys);
        MDB_txn* c; check(mdb_txn_begin(n.state_env, n.state_txn, 0, &c), "child begin");
        MDB_val kk{4, &key}, vv;
        std::uint64_t counter = 0;
        int rc = mdb_get(c, n.state_dbi, &kk, &vv);
        if (rc == 0) std::memcpy(&counter, vv.mv_data, 8); else if (rc != MDB_NOTFOUND) check(rc, "get");
        ++counter;
        MDB_val nv{8, &counter};
        check(mdb_put(c, n.state_dbi, &kk, &nv, 0), "state put");
        check(mdb_txn_commit(c), "child commit");
    }
}

static void begin_state(node& n) { check(mdb_txn_begin(n.state_env, nullptr, 0, &n.state_txn), "state begin"); }
static void commit_state(node& n) { check(mdb_txn_commit(n.state_txn), "state commit"); n.state_txn = nullptr; }
static void abort_state(node& n) { mdb_txn_abort(n.state_txn); n.state_txn = nullptr; }

int main(int, char** argv)
{
    const int arm = ARM;
    const double rate = std::strtod(argv[1], nullptr);
    const auto rtt = std::chrono::microseconds(std::strtoll(argv[2], nullptr, 10));
    const double seconds = std::strtod(argv[3], nullptr);
    const std::string dir = argv[4];
    node n;
    n.log_env = open_env(dir + "/log");
    n.state_env = open_env(dir + "/state");
    {
        MDB_txn* t; check(mdb_txn_begin(n.log_env, nullptr, 0, &t), "b"); check(mdb_dbi_open(t, nullptr, 0, &n.log_dbi), "dbi"); check(mdb_txn_commit(t), "c");
        check(mdb_txn_begin(n.state_env, nullptr, 0, &t), "b"); check(mdb_dbi_open(t, nullptr, 0, &n.state_dbi), "dbi"); check(mdb_txn_commit(t), "c");
    }
    const auto start = clk::now();
    const auto deadline = start + std::chrono::duration_cast<clk::duration>(std::chrono::duration<double>(seconds));
    auto arrival = [&](std::uint64_t i) { return start + std::chrono::duration_cast<clk::duration>(std::chrono::duration<double>(i / rate)); };
    auto arrived = [&](clk::time_point t) { return static_cast<std::uint64_t>(std::chrono::duration<double>(t - start).count() * rate); };

    std::uint64_t taken = 0;
    std::deque<batch> executed;
    std::size_t decided_batches = 0;
    std::vector<double> latency_us;
    std::uint64_t replays = 0, commits = 0, batches = 0;
    auto reply = [&](const batch& b, clk::time_point now) {
        for (std::uint64_t i = 0; i < b.count; ++i)
            latency_us.push_back(std::chrono::duration<double, std::micro>(now - arrival(b.first + i)).count());
    };
    auto propose = [&](clk::time_point now) {
        std::uint64_t upto = arrived(now);
        batch b{taken, upto - taken, {}};
        taken = upto;
        append_log(n, b);
        if (!n.state_txn) begin_state(n);
        execute(n, b);
        b.ack = clk::now() + rtt;
        executed.push_back(b);
        ++batches;
    };

    while (clk::now() < deadline) {
        auto now = clk::now();
        std::size_t acked = decided_batches;
        while (acked < executed.size() && executed[acked].ack <= now) ++acked;
        if (acked > decided_batches) {
            if (arm == 2 && acked < executed.size()) {
                abort_state(n);
                begin_state(n);
                for (std::size_t i = 0; i < acked; ++i) execute(n, executed[i]);
                commit_state(n);
                ++commits; ++replays;
                auto t = clk::now();
                for (std::size_t i = 0; i < acked; ++i) reply(executed[i], t);
                executed.erase(executed.begin(), executed.begin() + acked);
                begin_state(n);
                for (auto& b : executed) execute(n, b);
                decided_batches = 0;
            } else if (arm != 3 || acked == executed.size()) {
                commit_state(n);
                ++commits;
                auto t = clk::now();
                for (std::size_t i = 0; i < acked; ++i) reply(executed[i], t);
                executed.erase(executed.begin(), executed.begin() + acked);
                decided_batches = 0;
            } else {
                decided_batches = acked;
            }
            continue;
        }
        bool work = arrived(now) > taken;
        bool room = arm == 1 ? executed.empty() : arm == 2 ? executed.size() < max_in_flight : executed.size() < window_batches;
        if (work && room) { propose(now); continue; }
        auto wake = deadline;
        if (decided_batches < executed.size()) wake = std::min(wake, executed[decided_batches].ack);
        if (room) wake = std::min(wake, arrival(taken + 1));
        std::this_thread::sleep_until(wake);
    }
    if (n.state_txn) abort_state(n);
    std::sort(latency_us.begin(), latency_us.end());
    double med = latency_us.empty() ? 0 : latency_us[latency_us.size() / 2];
    double p99 = latency_us.empty() ? 0 : latency_us[latency_us.size() * 99 / 100];
    std::printf("arm=%d rate=%.0f rtt_us=%lld replied=%zu offered=%llu throughput_per_s=%.0f latency_median_us=%.1f latency_p99_us=%.1f batches=%llu commits=%llu replays=%llu\n",
                arm, rate, static_cast<long long>(rtt.count()), latency_us.size(), static_cast<unsigned long long>(arrived(deadline)),
                latency_us.size() / seconds, med, p99, static_cast<unsigned long long>(batches), static_cast<unsigned long long>(commits),
                static_cast<unsigned long long>(replays));
    mdb_env_close(n.log_env);
    mdb_env_close(n.state_env);
}
