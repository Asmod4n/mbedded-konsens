# Bench: node::handle(const received&) with and without the copy of the body

This run replaces 2026-10-20-212437Z.txt, which ran as root against the
rules.

What: decode of one message and node::handle(const received&) at a
follower in (Follower, Accept), over a store that only counts the
payload bytes. Two cases: an AcceptDecide with 64 entries of 1 KiB
(large), and a Decide (decide). Each message has the next sequence
number and is encoded outside the timed part. A sample is 25 messages
(large) or 4000 messages (decide); the time per message is the sample
divided by its count. Each process takes 31 samples and prints their
median.

Arms:
- before: 281439d. node copies the body into a message.
- after: f68f139. node passes the body that cbor::ref holds by const
  reference.
Each arm is its own binary. The arms run alternately, 7 rounds, and the
order swaps each round. The table compares the medians of the 7 process
medians.

Run length: the A/A pair is the after arm and the same bench.cpp
compiled a second time against the same library (after2). Max |A/A| of
the 6 cells: 31 samples 1.9 %, 101 samples 1.1 %. The run takes 31
samples. Noise floor: 1.9 %.

Build: Release, -std=c++23 -O2 -DNDEBUG, no -march (x86-64 baseline);
libkonsens.a built by its CMake with the same compiler, mbedded-cbor at
55418aa. Compilers: g++-15 (Ubuntu 15.3.0-4ubuntu1), g++-16 (Ubuntu
16.2.0-3ubuntu1), clang++-22 (Ubuntu clang 22.1.8-1ubuntu1). At run
time all use the libstdc++ of GCC 16 (the g++-15 binary needs
GLIBCXX_3.4.32, the others 3.4.36) and glibc (symbols up to GLIBC_2.38).
Machine: Firecracker VM, kernel 6.18.44-fc-v140, EEVDF scheduler (no
sched_ext), 4 cpus, Intel(R) Xeon(R) Processor, cpu flags sha256
733bd6f8a6de28df. Container: podman 4.9.3, Ubuntu 26.10 (development
branch), image localhost/cxx-matrix.
User and priority: each bench process runs as the user bench (uid
1001, not root), started by setpriv, at nice 0. The column
other_processes_of_user counts the other processes of that user before
each run: 0 in every row, so the run had priority over everything else
the user owns. Not pinned. Threads: 1. Three cpus stayed free. Steal

#include <algorithm>
#include <chrono>
#include <cstddef>
#include <cstdint>
#include <cstdlib>
#include <expected>
#include <format>
#include <iostream>
#include <span>
#include <string>
#include <string_view>
#include <system_error>
#include <variant>
#include <vector>

#include "konsens/node.hpp"
#include "konsens/wire.hpp"

namespace {

struct counting_storage {
    std::uint64_t log_len{};
    konsens::ballot promise{};
    std::uint64_t decided_idx{};
    konsens::ballot accepted_round{};
    std::uint64_t* payload_bytes;

    template <typename R>
    void count(const R& entries)
    {
        for (const auto& e : entries) {
            *payload_bytes += e.payload.size();
        }
    }

    std::expected<void, std::error_code> write_atomically(std::span<const konsens::storage_op> ops)
    {
        for (const auto& op : ops) {
            if (const auto* a = std::get_if<konsens::append_entries>(&op)) {
                count(a->entries);
                log_len += a->entries.size();
            } else if (const auto* p = std::get_if<konsens::append_on_prefix>(&op)) {
                count(p->entries);
                log_len = std::min(log_len, p->from_idx) + p->entries.size();
            } else if (const auto* s = std::get_if<konsens::set_promise>(&op)) {
                promise = s->promise;
            } else if (const auto* d = std::get_if<konsens::set_decided_index>(&op)) {
                decided_idx = d->decided_idx;
            } else if (const auto* r = std::get_if<konsens::set_accepted_round>(&op)) {
                accepted_round = r->accepted_round;
            }
        }
        return {};
    }
    std::expected<std::vector<konsens::entry>, std::error_code> get_entries(std::uint64_t from, std::uint64_t to) const
    {
        const auto end = std::min(to, log_len);
        return std::vector<konsens::entry>(from < end ? end - from : 0, konsens::entry{konsens::entry_kind::normal});
    }
    std::expected<std::uint64_t, std::error_code> get_log_len() const { return log_len; }
    std::expected<konsens::ballot, std::error_code> get_promise() const { return promise; }
    std::expected<std::uint64_t, std::error_code> get_decided_idx() const { return decided_idx; }
    std::expected<konsens::ballot, std::error_code> get_accepted_round() const { return accepted_round; }
};

}

#include "echo.hpp"

int main(int argc, char** argv)
{
    if (argc != 4) {
        std::cerr << "usage: bench large|decide <messages per sample> <samples>\n";
        return 2;
    }
    const std::string_view mode = argv[1];
    const auto per_sample = std::strtoull(argv[2], nullptr, 10);
    const auto samples = std::strtoull(argv[3], nullptr, 10);
    constexpr std::size_t entries_per_message = 64;
    constexpr std::size_t payload_size = 1024;
    constexpr std::uint64_t election_rounds = 40;

    using node_t = konsens::node<counting_storage, echo>;
    std::array<std::uint64_t, 3> bytes{};
    std::vector<node_t> nodes;
    for (konsens::node_id p = 1; p <= 3; ++p) {
        std::vector<konsens::node_id> peers;
        for (konsens::node_id q = 1; q <= 3; ++q) {
            if (q != p) {
                peers.push_back(q);
            }
        }
        nodes.push_back(*node_t::with({1, p, peers, 0, 3, 2}, counting_storage{.payload_bytes = &bytes.at(p - 1)},
                                      echo{}, 0));
    }
    const auto deliver = [&nodes] {
        for (bool more = true; more;) {
            more = false;
            for (auto& n : nodes) {
                for (const auto& m : n.take_outgoing_messages()) {
                    more = true;
                    const auto to = std::visit([](const auto& x) { return x.to; }, m);
                    if (!nodes.at(to - 1).handle(m)) {
                        std::abort();
                    }
                }
            }
        }
    };
    for (std::uint64_t i = 0; i < election_rounds; ++i) {
        deliver();
        for (auto& n : nodes) {
            if (!n.tick()) {
                std::abort();
            }
        }
    }
    deliver();
    const konsens::node_id leader = nodes.front().get_sequence_paxos().get_promise().pid;
    const konsens::node_id follower = leader == 1 ? 2 : 1;
    if (!nodes.at(leader - 1).append(konsens::entry{konsens::entry_kind::normal, {std::byte{1}}})) {
        std::abort();
    }
    konsens::accept_decide first{};
    for (const auto& m : nodes.at(leader - 1).take_outgoing_messages()) {
        if (const auto* p = std::get_if<konsens::paxos_message>(&m); p != nullptr && p->to == follower) {
            if (const auto* a = std::get_if<konsens::accept_decide>(&p->msg)) {
                first = *a;
            }
        }
    }
    auto& f = nodes.at(follower - 1);
    {
        auto b = konsens::encode(konsens::message{konsens::paxos_message{leader, follower, first}});
        if (!b || !f.handle(*konsens::decode(std::move(*b)))) {
            std::abort();
        }
    }
    static_cast<void>(f.take_outgoing_messages());
    const konsens::entry big{konsens::entry_kind::normal, std::vector<std::byte>(payload_size, std::byte{7})};
    const std::vector<konsens::entry> large(entries_per_message, big);
    std::uint64_t counter = first.seq_num.counter;
    std::vector<std::chrono::nanoseconds> times;
    for (std::uint64_t s = 0; s < samples; ++s) {
        std::vector<std::string> wire;
        wire.reserve(per_sample);
        for (std::uint64_t i = 0; i < per_sample; ++i) {
            counter += 1;
            const konsens::sequence_number seq{first.seq_num.session, counter};
            const konsens::paxos_msg body =
                mode == "large" ? konsens::paxos_msg{konsens::accept_decide{first.n, seq, first.decided_idx, large}}
                                : konsens::paxos_msg{konsens::decide{first.n, seq, first.decided_idx}};
            auto b = konsens::encode(konsens::message{konsens::paxos_message{leader, follower, body}});
            if (!b) {
                std::abort();
            }
            wire.push_back(std::move(*b));
        }
        const auto start = std::chrono::steady_clock::now();
        for (auto& w : wire) {
            auto r = konsens::decode(std::move(w));
            if (!r || !f.handle(*r)) {
                std::abort();
            }
        }
        times.push_back(std::chrono::steady_clock::now() - start);
        static_cast<void>(f.take_outgoing_messages());
    }
    std::ranges::sort(times);
    const auto median = times.at(times.size() / 2);
    std::cout << std::format("{} ns_per_message_median={} min={} max={} payload_bytes={}\n", mode,
                             median.count() / static_cast<long long>(per_sample),
                             times.front().count() / static_cast<long long>(per_sample),
                             times.back().count() / static_cast<long long>(per_sample), bytes.at(follower - 1));
}
