# containers in podman; builds with gcc of Ubuntu 24.04

## t.sql
pragma page_size=4096;
pragma journal_mode=wal;
create table log(idx integer primary key, record blob not null) strict;
create table kv(k text primary key, i integer, r any, t text, b blob) strict, without rowid;
with recursive n(x) as (select 1 union all select x+1 from n where x<2000)
insert into log select x, cast(printf('%089d', x) as blob) from n;
with recursive n(x) as (select 1 union all select x+1 from n where x<2000)
insert into kv select 'k'||x, x*7, x/3.0, printf('%08d',x), zeroblob(16) from n;
delete from kv where i % 5 = 0;
update kv set r = r*1.1 where i % 3 = 0;
create index kv_i on kv(i);
pragma wal_checkpoint(truncate);

## q.sql
select 'sum', sum(r) from kv;
select 'sum_ordered', sum(r order by k) from kv;
select 'avg', avg(r) from kv;
select 'total', total(r) from kv;
select 'text', cast(1.0/3 as text), cast(0.1+0.2 as text), 1e300*10, cast(2.0/3*1e20 as text);
select 'printf', printf('%.17g', 0.1+0.2), printf('%!.20g', 1.0/3);
select 'round', round(2.5), round(-2.5), round(0.125, 2);
select 'math', sin(1.0), pow(2.0, 0.5), exp(1.0), ln(10.0);
select 'int', 9223372036854775807 + 1, -9223372036854775808 - 1, 7/2, -7/2, 7%-3;
select 'json', json_object('a', 0.1, 'b', 1e100);
select 'order_noorder', group_concat(k) from (select k from kv where i < 100);
select 'like', 'Ä' like 'ä', upper('straße'), lower('ÄÖÜ'), length('😀'), unicode('😀');
select 'count', count(*), sum(i) from kv;
