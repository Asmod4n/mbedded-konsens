cmake_minimum_required(VERSION 3.25)
project(mbedded-konsens VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
find_package(doctest REQUIRED)
find_package(Boost REQUIRED CONFIG)

find_package(SQLite3 REQUIRED)

set(CBOR_INSTALL OFF)
add_subdirectory(deps/mbedded-cbor EXCLUDE_FROM_ALL)

block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS OFF)
    set(CRC32C_BUILD_TESTS OFF)
    set(CRC32C_BUILD_BENCHMARKS OFF)
    set(CRC32C_USE_GLOG OFF)
    set(CRC32C_INSTALL OFF)
    add_subdirectory(deps/crc32c EXCLUDE_FROM_ALL)
endblock()
set_target_properties(crc32c crc32c_sse42 crc32c_arm64 PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(KONSENS_CRC32C_INTERNAL_INCLUDES ${PROJECT_SOURCE_DIR}/deps/crc32c/src $<TARGET_PROPERTY:crc32c,BINARY_DIR>/include)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    set(KONSENS_HYDROGEN_TARGET_ARCH x86-64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(KONSENS_HYDROGEN_TARGET_ARCH armv8-a)
else()
    message(FATAL_ERROR "libhydrogen has no baseline -march for ${CMAKE_SYSTEM_PROCESSOR}")
endif()
block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS OFF)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
    set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
    set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
    set(HYDROGEN_TARGET_ARCH ${KONSENS_HYDROGEN_TARGET_ARCH})
    add_subdirectory(deps/libhydrogen EXCLUDE_FROM_ALL)
endblock()
set_target_properties(hydrogen-tests PROPERTIES EXCLUDE_FROM_ALL OFF)

set(KONSENS_WARNINGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)

file(GLOB KONSENS_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cpp)
add_library(konsens ${KONSENS_SOURCES})
target_include_directories(konsens PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(konsens PRIVATE SQLite::SQLite3)
target_link_libraries(konsens PRIVATE mbedded-cbor Boost::headers crc32c)
target_include_directories(konsens PRIVATE ${KONSENS_CRC32C_INTERNAL_INCLUDES})
target_include_directories(konsens PUBLIC $<TARGET_PROPERTY:mbedded-cbor,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_options(konsens PRIVATE ${KONSENS_WARNINGS})

enable_testing()
file(GLOB KONSENS_TESTS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/test/*.cpp)
foreach(test_source ${KONSENS_TESTS})
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE konsens doctest::doctest)
    target_compile_options(${test_name} PRIVATE ${KONSENS_WARNINGS})
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

file(GLOB KONSENS_EXAMPLES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/examples/*.cpp)
foreach(example_source ${KONSENS_EXAMPLES})
    get_filename_component(example_name ${example_source} NAME_WE)
    add_executable(${example_name} ${example_source})
    target_link_libraries(${example_name} PRIVATE konsens)
    target_compile_options(${example_name} PRIVATE ${KONSENS_WARNINGS})
endforeach()

target_link_libraries(test_wire PRIVATE mbedded-cbor)
target_link_libraries(test_crc32c PRIVATE crc32c)
target_link_libraries(test_verdict PRIVATE crc32c)
target_include_directories(test_crc32c PRIVATE ${KONSENS_CRC32C_INTERNAL_INCLUDES})
target_link_libraries(test_persistent_storage PRIVATE crc32c SQLite::SQLite3)
target_include_directories(test_persistent_storage PRIVATE ${PROJECT_SOURCE_DIR}/src)
find_program(KONSENS_SQLITE3_SHELL sqlite3 REQUIRED)
target_compile_definitions(test_persistent_storage PRIVATE "KONSENS_SQLITE3_SHELL=\"${KONSENS_SQLITE3_SHELL}\"")
target_link_libraries(test_sequence_paxos PRIVATE crc32c SQLite::SQLite3)
target_include_directories(test_sequence_paxos PRIVATE ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(test_node PRIVATE crc32c SQLite::SQLite3)
target_include_directories(test_node PRIVATE ${PROJECT_SOURCE_DIR}/src)

add_test(NAME counter_stress COMMAND counter_stress --clients 2 --increments 50 --allow-root-for-test)
add_test(NAME counter_help COMMAND counter --help)
add_test(NAME three_nodes_help COMMAND three_nodes --help)
add_test(NAME counter_stress_help COMMAND counter_stress --help)
add_test(NAME counter_stress_cbor COMMAND counter_stress --clients 2 --increments 50 --cbor --allow-root-for-test)

option(KONSENS_FUSE_TESTS "Build and run the tests of a failed fsync and a failed read on a FUSE file system" OFF)
if(KONSENS_FUSE_TESTS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
    find_program(KONSENS_FUSERMOUNT fusermount3 REQUIRED)
    add_executable(error_fs test/fuse/error_fs.cpp)
    target_link_libraries(error_fs PRIVATE PkgConfig::FUSE3)
    target_compile_definitions(error_fs PRIVATE FUSE_USE_VERSION=317)
    target_compile_options(error_fs PRIVATE ${KONSENS_WARNINGS})
    add_executable(test_device_faults test/fuse/test_device_faults.cpp)
    target_link_libraries(test_device_faults PRIVATE konsens doctest::doctest SQLite::SQLite3)
    target_include_directories(test_device_faults PRIVATE ${PROJECT_SOURCE_DIR}/test)
    target_compile_options(test_device_faults PRIVATE ${KONSENS_WARNINGS})
    target_compile_definitions(test_device_faults PRIVATE "KONSENS_ERROR_FS=\"$<TARGET_FILE:error_fs>\""
                                                          "KONSENS_FUSERMOUNT=\"${KONSENS_FUSERMOUNT}\"")
    add_dependencies(test_device_faults error_fs)
    add_test(NAME test_device_faults COMMAND test_device_faults)
    set_tests_properties(test_device_faults PROPERTIES TIMEOUT 60)
endif()

option(KONSENS_BENCH "Build konsens_bench and its A/A copy konsens_bench_aa, and test each case once" ON)
if(KONSENS_BENCH)
    foreach(bench_name konsens_bench konsens_bench_aa)
        add_executable(${bench_name} ${PROJECT_SOURCE_DIR}/bench/konsens_bench.cpp)
        target_link_libraries(${bench_name} PRIVATE konsens crc32c SQLite::SQLite3 Threads::Threads)
        target_include_directories(${bench_name} PRIVATE ${PROJECT_SOURCE_DIR}/src)
        target_compile_options(${bench_name} PRIVATE ${KONSENS_WARNINGS})
    endforeach()
    add_test(NAME konsens_bench_each_case_once COMMAND konsens_bench --each-case-once --allow-root-for-test)
    set_tests_properties(konsens_bench_each_case_once PROPERTIES TIMEOUT 900)
endif()

set(KONSENS_TEST_TMPDIR "" CACHE PATH "The directory of the stores of the tests, a tmpfs; empty keeps TMPDIR")
if(KONSENS_TEST_TMPDIR)
    get_property(konsens_all_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${konsens_all_tests} PROPERTIES ENVIRONMENT "TMPDIR=${KONSENS_TEST_TMPDIR}")
endif()


option(KONSENS_PROGRAM "Build konsensd, the program that runs one node over TCP and HTTP" OFF)
if(KONSENS_PROGRAM)
    include(ExternalProject)
    find_package(tomlplusplus REQUIRED)
    find_package(ada REQUIRED)
    set(KONSENS_SLIPSTREAM ${PROJECT_SOURCE_DIR}/deps/slipstreamIO)
    set(KONSENS_LIBURING_TREE ${CMAKE_BINARY_DIR}/liburing)
    set(KONSENS_LIBURING_OUT ${CMAKE_BINARY_DIR}/liburing-out)
    ExternalProject_Add(carried_liburing
        SOURCE_DIR ${KONSENS_LIBURING_TREE}
        DOWNLOAD_COMMAND ${CMAKE_COMMAND} -E copy_directory ${KONSENS_SLIPSTREAM}/deps/liburing ${KONSENS_LIBURING_TREE}
        PATCH_COMMAND ${CMAKE_COMMAND} -E copy ${KONSENS_SLIPSTREAM}/src/liburing_syscall.h ${KONSENS_LIBURING_TREE}/src/syscall.h
              COMMAND ${CMAKE_COMMAND} -E copy ${KONSENS_SLIPSTREAM}/src/liburing_arch_syscall.h
                      ${KONSENS_SLIPSTREAM}/src/slipstream_syscall.h ${KONSENS_LIBURING_TREE}/src/
        CONFIGURE_COMMAND ./configure --prefix=${KONSENS_LIBURING_OUT} --cc=${CMAKE_C_COMPILER} --cxx=${CMAKE_CXX_COMPILER}
        BUILD_COMMAND make -C src ENABLE_SHARED=0 liburing.a
        INSTALL_COMMAND make -C src ENABLE_SHARED=0 install
        BUILD_IN_SOURCE ON
        BUILD_BYPRODUCTS ${KONSENS_LIBURING_OUT}/lib/liburing.a)
    file(MAKE_DIRECTORY ${KONSENS_LIBURING_OUT}/include)
    file(GLOB KONSENS_SLIPSTREAM_ENGINE ${KONSENS_SLIPSTREAM}/src/engine_*.c)
    add_library(slipstream STATIC ${KONSENS_SLIPSTREAM}/src/slipstream_syscall.c ${KONSENS_SLIPSTREAM}/src/slipstream_engine.c
                                  ${KONSENS_SLIPSTREAM}/src/slipstream_signal.c ${KONSENS_SLIPSTREAM_ENGINE})
    target_include_directories(slipstream SYSTEM PUBLIC ${KONSENS_SLIPSTREAM}/src ${KONSENS_LIBURING_OUT}/include)
    add_dependencies(slipstream carried_liburing)
    target_link_libraries(slipstream PUBLIC Threads::Threads)

    file(GLOB KONSENS_PROGRAM_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/program/*.cpp)
    list(FILTER KONSENS_PROGRAM_SOURCES EXCLUDE REGEX "/main\\.cpp$")
    add_library(konsensd_parts STATIC ${KONSENS_PROGRAM_SOURCES})
    target_include_directories(konsensd_parts PUBLIC ${PROJECT_SOURCE_DIR}/program)
    target_include_directories(konsensd_parts SYSTEM PUBLIC ${PROJECT_SOURCE_DIR}/deps/mruby-webmachine/src)
    target_link_libraries(konsensd_parts PUBLIC konsens mbedded-cbor ada::ada
                                                ${KONSENS_LIBURING_OUT}/lib/liburing.a slipstream)
    target_compile_options(konsensd_parts PRIVATE ${KONSENS_WARNINGS})
    add_executable(konsensd ${PROJECT_SOURCE_DIR}/program/main.cpp)
    target_link_libraries(konsensd PRIVATE konsensd_parts)
    target_compile_options(konsensd PRIVATE ${KONSENS_WARNINGS})

    file(GLOB KONSENS_PROGRAM_TESTS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/program/test/*.cpp)
    foreach(test_source ${KONSENS_PROGRAM_TESTS})
        get_filename_component(test_name ${test_source} NAME_WE)
        add_executable(${test_name} ${test_source})
        target_link_libraries(${test_name} PRIVATE konsensd_parts doctest::doctest)
        target_compile_options(${test_name} PRIVATE ${KONSENS_WARNINGS})
        add_test(NAME ${test_name} COMMAND ${test_name})
    endforeach()
endif()
