cmake_minimum_required(VERSION 3.25)
project(mbedded-konsens VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)
find_package(doctest REQUIRED)
find_package(Boost REQUIRED CONFIG)

set(KONSENS_LMDB_SOURCE_DIR ${PROJECT_SOURCE_DIR}/deps/lmdb/libraries/liblmdb)
add_library(lmdb STATIC ${KONSENS_LMDB_SOURCE_DIR}/mdb.c ${KONSENS_LMDB_SOURCE_DIR}/midl.c
            ${KONSENS_LMDB_SOURCE_DIR}/module.c)
target_include_directories(lmdb PUBLIC ${KONSENS_LMDB_SOURCE_DIR})
set(KONSENS_LMDB_WARNINGS -W -Wall -Wno-unused-parameter -Wbad-function-cast -Wuninitialized)
target_compile_options(lmdb PRIVATE ${KONSENS_LMDB_WARNINGS})
target_link_libraries(lmdb PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(lmdb PROPERTIES POSITION_INDEPENDENT_CODE ON)

set(KONSENS_LMDB_0_9_SOURCE_DIR ${PROJECT_SOURCE_DIR}/deps/lmdb-0.9/libraries/liblmdb)
add_library(lmdb-0.9 STATIC ${KONSENS_LMDB_0_9_SOURCE_DIR}/mdb.c ${KONSENS_LMDB_0_9_SOURCE_DIR}/midl.c)
target_include_directories(lmdb-0.9 PUBLIC ${KONSENS_LMDB_0_9_SOURCE_DIR})
target_compile_options(lmdb-0.9 PRIVATE ${KONSENS_LMDB_WARNINGS})
target_link_libraries(lmdb-0.9 PUBLIC Threads::Threads)

foreach(program mdb_dump mdb_load)
    add_executable(${program} ${KONSENS_LMDB_SOURCE_DIR}/${program}.c)
    target_compile_options(${program} PRIVATE ${KONSENS_LMDB_WARNINGS})
    target_link_libraries(${program} PRIVATE lmdb)
    add_executable(${program}-0.9 ${KONSENS_LMDB_0_9_SOURCE_DIR}/${program}.c)
    target_compile_options(${program}-0.9 PRIVATE ${KONSENS_LMDB_WARNINGS})
    target_link_libraries(${program}-0.9 PRIVATE lmdb-0.9)
endforeach()

set(CBOR_INSTALL OFF)
add_subdirectory(deps/mbedded-cbor EXCLUDE_FROM_ALL)

block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS OFF)
    set(CRC32C_BUILD_TESTS OFF)
    set(CRC32C_BUILD_BENCHMARKS OFF)
    set(CRC32C_USE_GLOG OFF)
    set(CRC32C_INSTALL OFF)
    add_subdirectory(deps/crc32c EXCLUDE_FROM_ALL)
endblock()
set_target_properties(crc32c crc32c_sse42 crc32c_arm64 PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(KONSENS_CRC32C_INTERNAL_INCLUDES ${PROJECT_SOURCE_DIR}/deps/crc32c/src $<TARGET_PROPERTY:crc32c,BINARY_DIR>/include)

if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    set(KONSENS_HYDROGEN_TARGET_ARCH x86-64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(KONSENS_HYDROGEN_TARGET_ARCH armv8-a)
else()
    message(FATAL_ERROR "libhydrogen has no baseline -march for ${CMAKE_SYSTEM_PROCESSOR}")
endif()
block(SCOPE_FOR VARIABLES)
    set(BUILD_SHARED_LIBS OFF)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
    set(CMAKE_POLICY_DEFAULT_CMP0126 NEW)
    set(CMAKE_EXPORT_NO_PACKAGE_REGISTRY ON)
    set(HYDROGEN_TARGET_ARCH ${KONSENS_HYDROGEN_TARGET_ARCH})
    add_subdirectory(deps/libhydrogen EXCLUDE_FROM_ALL)
endblock()
set_target_properties(hydrogen-tests PROPERTIES EXCLUDE_FROM_ALL OFF)

set(KONSENS_WARNINGS -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)

file(GLOB KONSENS_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cpp)
add_library(konsens ${KONSENS_SOURCES})
target_include_directories(konsens PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(konsens PRIVATE lmdb)
target_link_libraries(konsens PRIVATE mbedded-cbor Boost::headers crc32c)
target_include_directories(konsens PRIVATE ${KONSENS_CRC32C_INTERNAL_INCLUDES})
target_include_directories(konsens PUBLIC $<TARGET_PROPERTY:mbedded-cbor,INTERFACE_INCLUDE_DIRECTORIES>)
target_compile_options(konsens PRIVATE ${KONSENS_WARNINGS})

enable_testing()
file(GLOB KONSENS_TESTS CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/test/*.cpp)
foreach(test_source ${KONSENS_TESTS})
    get_filename_component(test_name ${test_source} NAME_WE)
    add_executable(${test_name} ${test_source})
    target_link_libraries(${test_name} PRIVATE konsens doctest::doctest)
    target_compile_options(${test_name} PRIVATE ${KONSENS_WARNINGS})
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()

file(GLOB KONSENS_EXAMPLES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/examples/*.cpp)
foreach(example_source ${KONSENS_EXAMPLES})
    get_filename_component(example_name ${example_source} NAME_WE)
    add_executable(${example_name} ${example_source})
    target_link_libraries(${example_name} PRIVATE konsens)
    target_compile_options(${example_name} PRIVATE ${KONSENS_WARNINGS})
endforeach()

target_link_libraries(test_wire PRIVATE mbedded-cbor)
target_link_libraries(test_crc32c PRIVATE crc32c)
target_include_directories(test_crc32c PRIVATE ${KONSENS_CRC32C_INTERNAL_INCLUDES})
target_link_libraries(test_persistent_storage PRIVATE crc32c)
add_dependencies(test_lmdb_upgrade mdb_dump mdb_load mdb_dump-0.9 mdb_load-0.9)
target_compile_definitions(
    test_lmdb_upgrade PRIVATE "KONSENS_MDB_DUMP=\"$<TARGET_FILE:mdb_dump>\"" "KONSENS_MDB_LOAD=\"$<TARGET_FILE:mdb_load>\""
    "KONSENS_MDB_DUMP_0_9=\"$<TARGET_FILE:mdb_dump-0.9>\"" "KONSENS_MDB_LOAD_0_9=\"$<TARGET_FILE:mdb_load-0.9>\""
    "KONSENS_LSAN_MDB_DUMP_0_9=\"${PROJECT_SOURCE_DIR}/test/lsan-mdb_dump-0.9.txt\"")

add_test(NAME counter_stress COMMAND counter_stress --clients 2 --increments 50 --allow-root-for-test)
add_test(NAME counter_stress_cbor COMMAND counter_stress --clients 2 --increments 50 --cbor --allow-root-for-test)

option(KONSENS_FUSE_TESTS "Build and run the tests of a failed fsync and a failed read on a FUSE file system" OFF)
if(KONSENS_FUSE_TESTS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
    find_program(KONSENS_FUSERMOUNT fusermount3 REQUIRED)
    add_executable(error_fs test/fuse/error_fs.cpp)
    target_link_libraries(error_fs PRIVATE PkgConfig::FUSE3)
    target_compile_definitions(error_fs PRIVATE FUSE_USE_VERSION=317)
    target_compile_options(error_fs PRIVATE ${KONSENS_WARNINGS})
    add_executable(test_device_faults test/fuse/test_device_faults.cpp)
    target_link_libraries(test_device_faults PRIVATE konsens doctest::doctest)
    target_include_directories(test_device_faults PRIVATE ${PROJECT_SOURCE_DIR}/test)
    target_compile_options(test_device_faults PRIVATE ${KONSENS_WARNINGS})
    target_compile_definitions(test_device_faults PRIVATE "KONSENS_ERROR_FS=\"$<TARGET_FILE:error_fs>\""
                                                          "KONSENS_FUSERMOUNT=\"${KONSENS_FUSERMOUNT}\"")
    add_dependencies(test_device_faults error_fs)
    add_test(NAME test_device_faults COMMAND test_device_faults)
    set_tests_properties(test_device_faults PROPERTIES TIMEOUT 60)
endif()

set(KONSENS_TEST_TMPDIR "" CACHE PATH "The directory of the LMDB environments of the tests, a tmpfs; empty keeps TMPDIR")
if(KONSENS_TEST_TMPDIR)
    get_property(konsens_all_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${konsens_all_tests} PROPERTIES ENVIRONMENT "TMPDIR=${KONSENS_TEST_TMPDIR}")
endif()

option(KONSENS_BENCH "Build konsens_bench and its A/A copy konsens_bench_aa" OFF)
if(KONSENS_BENCH)
    foreach(bench_name konsens_bench konsens_bench_aa)
        add_executable(${bench_name} ${PROJECT_SOURCE_DIR}/bench/konsens_bench.cpp)
        target_link_libraries(${bench_name} PRIVATE konsens crc32c lmdb Threads::Threads)
        target_include_directories(${bench_name} PRIVATE ${PROJECT_SOURCE_DIR}/src)
        target_compile_options(${bench_name} PRIVATE ${KONSENS_WARNINGS})
    endforeach()
endif()
